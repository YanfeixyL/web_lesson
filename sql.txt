CREATE DATABASE IF NOT EXISTS haobangshou DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE haobangshou;
SET FOREIGN_KEY_CHECKS=0;
DROP TABLE IF EXISTS successes, responses, help_requests, regions, users;
CREATE TABLE users (
  id INT PRIMARY KEY AUTO_INCREMENT, username VARCHAR(30) NOT NULL UNIQUE,
  password_hash VARCHAR(100) NOT NULL, role ENUM('admin','user') NOT NULL DEFAULT 'user',
  name VARCHAR(30) NOT NULL, phone CHAR(11) NOT NULL, bio VARCHAR(500) DEFAULT '',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);
CREATE TABLE regions (
  id INT PRIMARY KEY AUTO_INCREMENT, name VARCHAR(50) NOT NULL, city VARCHAR(50) NOT NULL, province VARCHAR(50) NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);
CREATE TABLE help_requests (
  id INT PRIMARY KEY AUTO_INCREMENT, user_id INT NOT NULL, region_id INT NOT NULL, type VARCHAR(30) NOT NULL,
  title VARCHAR(100) NOT NULL, description TEXT NOT NULL, address VARCHAR(200) NOT NULL, attachment VARCHAR(255) DEFAULT '', status TINYINT NOT NULL DEFAULT 0,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_request_user FOREIGN KEY(user_id) REFERENCES users(id), CONSTRAINT fk_request_region FOREIGN KEY(region_id) REFERENCES regions(id)
);
CREATE TABLE responses (
  id INT PRIMARY KEY AUTO_INCREMENT, request_id INT NOT NULL, user_id INT NOT NULL, description TEXT NOT NULL, attachment VARCHAR(255) DEFAULT '', status TINYINT NOT NULL DEFAULT 0,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_response_request FOREIGN KEY(request_id) REFERENCES help_requests(id), CONSTRAINT fk_response_user FOREIGN KEY(user_id) REFERENCES users(id)
);
CREATE TABLE successes (
  id INT PRIMARY KEY AUTO_INCREMENT, request_id INT NOT NULL, response_id INT NOT NULL, accepted_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_success_request FOREIGN KEY(request_id) REFERENCES help_requests(id), CONSTRAINT fk_success_response FOREIGN KEY(response_id) REFERENCES responses(id)
);
